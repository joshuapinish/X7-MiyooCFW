cmake_minimum_required(VERSION 3.16)
project(X7MiyooCFWPort VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(X7_RAM_MIB "32" CACHE STRING "Configured RAM size in MiB (32..256)")
if(NOT X7_RAM_MIB MATCHES "^[0-9]+$")
    message(FATAL_ERROR "X7_RAM_MIB must be an integer between 32 and 256")
endif()
if(X7_RAM_MIB LESS 32 OR X7_RAM_MIB GREATER 256)
    message(FATAL_ERROR "X7_RAM_MIB=${X7_RAM_MIB} is outside the supported 32..256 MiB range")
endif()
option(X7_BUILD_TESTS "Build host-side unit tests" ON)
add_library(x7display STATIC src/display/display_bus.c src/display/panel_hsd045f8w7.c src/platform/memory_profile.c)
target_include_directories(x7display PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(x7display PUBLIC X7_RAM_MIB=${X7_RAM_MIB})
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(x7display PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(X7_BUILD_TESTS)
    enable_testing()
    add_executable(test_display tests/test_display.c)
    target_link_libraries(test_display PRIVATE x7display)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_display PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME display_core COMMAND test_display)
    add_executable(test_memory_profile tests/test_memory_profile.c)
    target_link_libraries(test_memory_profile PRIVATE x7display)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_memory_profile PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME memory_profile COMMAND test_memory_profile)
endif()
